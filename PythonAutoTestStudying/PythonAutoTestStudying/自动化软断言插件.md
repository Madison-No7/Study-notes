import pytest_check as check

def LoginedTest(self):
    writer_name = self.wait.until(EC.visibility_of_element_located(
        (By.CSS_SELECTOR, 'body > div.container > div.left > div > h3')
    ))
    writer_name_text = writer_name.text
    ChromeDriver.getscreenshot()

    # 软断言：失败时记录错误并高亮报告，但不会中断后续代码执行
    check.equal(writer_name_text, "zhangsan", "作者昵称不一致")

    # 后续代码继续正常执行
    article_title = self.wait.until(EC.visibility_of_element_located(
        (By.CSS_SELECTOR, "body > div.container > div.right > div:nth-child(1) > div.title")
    ))
    article_title_text = article_title.text
    check.equal(article_title_text, "日本战败投降81周年", "文章标题不一致")
